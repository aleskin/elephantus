# Конфиг игры Elephantus.
#
# Секции [circus.…] — мини-игра «Цирковой слоник». У каждой мини-игры свои секции
# с её именем впереди; [pad] — общая для всей игры.
#
# Меняй числа, сохраняй файл — игра подхватит их сама примерно через секунду,
# перезапускать не нужно. Если в файле ошибка, игра покажет её внизу экрана
# и продолжит играть со старыми числами.
#
# Дробные числа можно писать и через точку, и через запятую: 1.5 или 1,5.
# Все числа должны быть больше нуля. Строку можно удалить — тогда берётся
# число из кода (оно сейчас такое же, как здесь).
# Пока числа временные: подбираем их вместе с Полиной (задача CEL-00031).

[circus.elephant]
# Скорость слоника на двух ногах, метров в секунду.
# Больше — быстрее идёт. От старта до обруча 12 метров.
walk_speed = 3

# Высота прыжка с сальтухой, метров. Середина обруча — на высоте 1,6 метра.
jump_height = 1.3

# Тяжесть: насколько быстро слоник падает вниз после прыжка.
# Больше — прыжок короче и резче, меньше — слоник дольше висит в воздухе.
gravity = 20

# Сколько секунд слоник может идти без кнопки A, пока не встанет на четыре ноги.
fall_grace = 0.3

[circus.buttons]
# «Очень быстро»: самая длинная пауза между X и B для прыжка, в секундах.
# Больше — легче попасть.
salto_window = 0.4

[circus.game]
# Пауза после падения или касания обруча, пока уровень не начнётся сначала, секунд.
restart_delay = 1
# Очки за сальтуху — вперёд или назад, даже без обруча.
salto_points = 5
# Сальтуха сквозь обруч: очки умножаются на это число.
hoop_multiplier = 2
# Сколько попыток в одной игре: упал или задел обруч — минус попытка.
attempts = 3

[pad]
# Джойстик: какая кнопка где.
# nintendo — как говорит Полина: A справа, B снизу, X сверху, Y слева.
#            На джойстике в форме PlayStation (Xtrike Me GP-43): A = ○, B = ✕, X = △, Y = □.
# xbox     — для джойстика с буквами Xbox: A снизу, B справа, X слева, Y сверху.
layout = nintendo
